$(document).ready(function () { $('#customerId').keypress(function (e) { // e.preventDefault(); e.stopImmediatePropagation(); var keycode = (event.keyCode ? event.keyCode : event.which); if (canAdd == 1) { if (keycode == '13') { if ($("#customerId").val() != '') { getDataCustomer($("#customerId").val()); } else {} } } }) $('#btnFindcustomerId').click(function (e) { e.preventDefault(); e.stopImmediatePropagation(); $('#dxGridModalCustomer').modal('show'); cariCustomer(event) }) }); function getDataCustomer(e) { $.ajax({ url: host + "/getDataCustomer", method: "GET", data: { "customerId": e, }, dataType: "json", success: function (datas) { if (datas.length != 0) { $('#customerId').val(datas[0].CUSID); $('#namaPerusahaan').val(datas[0].NAME); $('#idKlpPasien').val(datas[0].FMKCUST_ID); $('#namaKlpPasien').val(datas[0].FMKCUSTN); $('#namaAsuransi').focus() } else { $('#dxGridModalCustomer').modal('show'); cariCustomer() } } }) } function cariCustomer() { $("#dxGridDataCustomer").dxDataGrid({ showBorders: true, showRowLines: true, columnAutoWidth: true, showColumnLines: false, hoverStateEnabled: true, rowAlternationEnabled: true, paging: { pageSize: 1000 }, filterRow: { visible: false }, searchPanel: { visible: true }, dataSource: new DevExpress.data.DataSource({ load: function (e) { var deferred = $.Deferred(); $.ajax({ url: host + "/getCustomer", method: "GET", data: { "idKlpPasien": $('#idKlpPasien').val(), }, success: function (result) { deferred.resolve(result, { totalCount: result.length }); } }); return deferred.promise(); } }), selection: { mode: "single" }, columns: [{ dataField: "CUSID", caption: "Kode Customer", }, { dataField: "NAME", caption: "Nama Customer", }, ], onRowClick: function (e) { $("#customerId").val(e.data.CUSID); $("#namaPerusahaan").val(e.data.NAME); $("#idKlpPasien").val(e.data.FMKCUST_ID); $("#namaKlpPasien").val(e.data.FMKCUSTN); $('#dxGridModalCustomer').modal('hide'); $('#namaAsuransi').focus() } }); }
