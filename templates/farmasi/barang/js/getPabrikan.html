$(document).ready(function () { $('#Findpabrikan').click(function (e) { gridFindpabrikan(e) }) $('.grid-opt-pabrikan').keypress(function (event) { var keycode = (event.keyCode ? event.keyCode : event.which); if (keycode == '13') { caripabrikan(event) } }); }) function gridFindpabrikan(e) { e.preventDefault(); e.stopImmediatePropagation(); caripabrikan(e) } var gridDatapabrikan = new DevExpress.data.DataSource({ load: function (values) { var search_kode_pabrikan = $('#search_kode_pabrikan').val(); var search_nama_pabrikan = $('#search_nama_pabrikan').val(); return $.ajax({ url: host + "/getpabrikan", method: "GET", data: { "search_kode_pabrikan": search_kode_pabrikan, "search_nama_pabrikan": search_nama_pabrikan }, }) } }); function caripabrikan(e, param = null) { $('#DxGridModalpabrikan').modal('show'); $('#DxGridLabelpabrikan').html('pabrikan'); //e.preventDefault(); e.stopImmediatePropagation(); $("#DxgridDatapabrikan2").dxDataGrid({ showColumnLines: false, showRowLines: true, rowAlternationEnabled: true, showBorders: true, paging: { pageSize: 1000 }, filterRow: { visible: false }, //searchPanel : { visible: true }, dataSource: gridDatapabrikan, selection: { mode: "single" }, hoverStateEnabled: true, columns: [{ dataField: "FMPSUPPLIERC", caption: "KODE", width: 220, }, { dataField: "FMPNAME_SUPPL", caption: "NAMA", } ], onRowClick: function (e) { if (param == null) { $("#pabrikan_barang").val(e.data.FMPSUPPLIERC) $("#pabrikann_barang").val(e.data.FMPNAME_SUPPL) } else if (param == '1') { $("#kodepabrikan_dari").val(e.data.FMPSUPPLIERC) } else if (param == '2') { $("#kodepabrikan_sampai").val(e.data.FMPSUPPLIERC) } $('#DxGridModalpabrikan').modal('hide'); } }); }
