$(document).ready(function () { $('#idKlpPasien').keypress(function (e) { // e.preventDefault(); e.stopImmediatePropagation(); var keycode = (event.keyCode ? event.keyCode : event.which); if (canAdd == 1) { if (keycode == '13') { if ($("#idKlpPasien").val() != '') { getDataKlpPasien($("#idKlpPasien").val()); } else {} } } }) $('#btnFindKlpPasien').click(function (e) { e.preventDefault(); e.stopImmediatePropagation(); $('#dxGridModalKelcustomer').modal('show'); cariKlpPasien(event) }) }); function getDataKlpPasien(e) { $.ajax({ url: host + "/getDataKlpPasien", method: "GET", data: { "idKlpPasien": e, }, dataType: "json", success: function (datas) { if (datas.length != 0) { $('#idKlpPasien').val(datas[0].FMKCUST_ID); $('#namaKlpPasien').val(datas[0].FMKCUSTN); $('#customerId').val(''); $('#namaPerusahaan').val(''); $('#customerId').focus() } else { $('#dxGridModalKelcustomer').modal('show'); cariKlpPasien() } } }) } function cariKlpPasien() { $("#dxGridDataKelcustomer").dxDataGrid({ showBorders: true, showRowLines: true, columnAutoWidth: true, showColumnLines: false, hoverStateEnabled: true, rowAlternationEnabled: true, paging: { pageSize: 1000 }, filterRow: { visible: false }, searchPanel: { visible: true }, dataSource: new DevExpress.data.DataSource({ load: function (e) { var deferred = $.Deferred(); $.ajax({ url: host + "/getKlpPasien", method: "GET", data: {}, success: function (result) { deferred.resolve(result, { totalCount: result.length }); } }); return deferred.promise(); } }), selection: { mode: "single" }, columns: [{ dataField: "FMKCUST_ID", caption: "Kode KlpPasien", }, { dataField: "FMKCUSTN", caption: "Nama KlpPasien", }, ], onRowClick: function (e) { $("#idKlpPasien").val(e.data.FMKCUST_ID); $("#namaKlpPasien").val(e.data.FMKCUSTN); $('#customerId').val(''); $('#namaPerusahaan').val(''); $('#dxGridModalKelcustomer').modal('hide'); $('#customerId').focus() } }); }
