return $.ajax({ url: host + "/getpoliklinik", method: "GET", data: { "search_kode_poliklinik": search_kode_poliklinik, "search_nama_poliklinik": search_nama_poliklinik }, success: function (datas) { $('#DxGridModalpoliklinik').modal('show'); } }) } }); function caripoliklinik(e, param = null) { console.log('menu cari clicked roti') //e.preventDefault(); e.stopImmediatePropagation(); $("#DxgridDatapoliklinik2").dxDataGrid({ showColumnLines: false, showRowLines: true, rowAlternationEnabled: true, showBorders: true, paging: { pageSize: 10000 }, filterRow: { visible: false }, //searchPanel : { visible: true }, dataSource: gridDatapoliklinik, selection: { mode: "single" }, hoverStateEnabled: true, columns: [{ dataField: "FMPKLINIK_ID", caption: "KODE", width: 220, }, { dataField: "FMPKLINIKN", caption: "NAMA", }, { dataField: "FMPKODEBPJS", caption: "Kode BPJS", }, { dataField: "SSFHIR_KD_POLI", caption: "Kode Satu Sehat", } ], onRowClick: function (e) { if (param == null) { $("#kode_poliklinik").val(e.data.FMPKLINIK_ID) $("#nama_poliklinik").val(e.data.FMPKLINIKN) $("#kode_klinik_bpjs").val(e.data.FMPKODEBPJS) $("#kode_klinik_satu_sehat").val(e.data.SSFHIR_KD_POLI) $('#DxGridModalpoliklinik').modal('hide'); $('#loading_title').html('Proses Cetak..'); $('#loading_text').html('Tunggu Sebentar'); }, complete: function () { $('#ModalLoading').modal('hide'); }, success: function (datas) { console.log(datas);
